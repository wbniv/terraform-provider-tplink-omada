func TestControllerLabSiteCredentials(t *testing.T) {
	if os.Getenv("OMADA_CONTROLLER_LAB") != "1" {
		t.Skip("disposable controller lab only")
	}
	if os.Getenv("OMADA_URL") != "https://controller:8043" {
		t.Fatal("refusing a target outside the isolated lab")
	}
	bootstrap := os.Getenv("OMADA_LAB_DEVICE_PASSWORD")
	if bootstrap == "" {
		t.Fatal("missing private lab device credential")
	}
	site := fmt.Sprintf(`
resource "omada_site" "lab" {
  name = "Terraform Credential Lab"
  region = "United States"
  time_zone = "UTC"
  scenario = "Office"
  device_account_username = "lab-device"
  device_account_password = %q
}
`, bootstrap)
	config := func(password string) string {
		return site + fmt.Sprintf(`
resource "omada_site_settings" "lab" {
  site_id = omada_site.lab.id
  device_account_username = "lab-rotated"
  device_account_password = %q
}
`, password)
	}
	var siteID string
	var preserved map[string]any
	check := func(password string) resource.TestCheckFunc {
		return func(state *terraform.State) error {
			if err := resource.TestCheckResourceAttr("omada_site_settings.lab", "device_account_password", password)(state); err != nil {
				return fmt.Errorf("configured credential not retained in state")
			}
			id := state.RootModule().Resources["omada_site.lab"].Primary.ID
			if siteID != "" && id != siteID {
				return fmt.Errorf("credential rotation replaced the site")
			}
			siteID = id
			c, err := client.NewClient(os.Getenv("OMADA_URL"), os.Getenv("OMADA_USERNAME"), os.Getenv("OMADA_PASSWORD"), true)
			if err != nil {
				return err
			}
			raw, err := c.GetSiteSettingsRaw(context.Background(), id)
			if err != nil {
				return err
			}
			account, ok := raw["deviceAccount"].(map[string]any)
			if !ok || account["username"] != "lab-rotated" {
				return fmt.Errorf("controller did not retain the rotated username")
			}
			current := map[string]any{}
			for _, section := range []string{"mesh", "roaming", "led", "lldp", "bandSteering"} {
				current[section] = raw[section]
			}
			if preserved != nil && !reflect.DeepEqual(preserved, current) {
				return fmt.Errorf("credential rotation changed unrelated settings")
			}
			preserved = current
			return nil
		}
	}
	resource.Test(t, resource.TestCase{
		PreCheck: func() { testAccPreCheck(t) }, ProtoV6ProviderFactories: testAccProtoV6ProviderFactories,
		Steps: []resource.TestStep{
			{Config: site, Check: resource.TestCheckResourceAttrSet("omada_site.lab", "id")},
			{Config: config(bootstrap + "R1"), Check: check(bootstrap + "R1")},
			{Config: config(bootstrap + "R2"), Check: check(bootstrap + "R2")},
			{Config: config(bootstrap + "R2"), PlanOnly: true},
		},
	})
}

func TestControllerLabWirelessLifecycle(t *testing.T) {
 if os.Getenv("OMADA_CONTROLLER_LAB") != "1" { t.Skip("disposable controller lab only") }
 if os.Getenv("OMADA_URL") != "https://controller:8043" { t.Fatal("refusing a target outside the isolated lab") }
 config := func(name, group string) string {
  return fmt.Sprintf(`
resource "omada_wlan_group" "lab" {
 site_id = %q
 name = "L1 Custom"
}
resource "omada_wireless_network" "lab" {
 site_id = %q
 name = %q
 security = 3
 passphrase = "L1-synthetic-test-password"
 pmf_mode = 2
 %s
}
`, os.Getenv("OMADA_TEST_SITE_ID"), os.Getenv("OMADA_TEST_SITE_ID"), name, group)
 }
 var preserved map[string]any
 observe := func(compare bool) resource.TestCheckFunc {
  return func(state *terraform.State) error {
   attrs := state.RootModule().Resources["omada_wireless_network.lab"].Primary.Attributes
   c, err := client.NewClient(os.Getenv("OMADA_URL"), os.Getenv("OMADA_USERNAME"), os.Getenv("OMADA_PASSWORD"), true)
   if err != nil { return err }
   raw, err := c.GetWirelessNetworkRaw(context.Background(), attrs["site_id"], attrs["wlan_group_id"], attrs["id"])
   if err != nil { return err }
   current := map[string]any{}
   for _, key := range []string{"deviceType", "vlanEnable", "vlanSetting", "rateLimit", "ssidRateLimit", "multiCastSetting", "rateAndBeaconCtrl", "mloEnable", "prohibitWifiShare", "guestNetEnable"} { current[key] = raw[key] }
   if compare && !reflect.DeepEqual(preserved, current) { return fmt.Errorf("SSID update changed unrelated settings") }
   preserved = current
   return nil
  }
 }
 custom := "wlan_group_id = omada_wlan_group.lab.id"
 resource.Test(t, resource.TestCase{
  PreCheck: func() { testAccPreCheck(t) }, ProtoV6ProviderFactories: testAccProtoV6ProviderFactories,
  Steps: []resource.TestStep{
   {Config: config("L1 Initial", ""), Check: resource.TestCheckResourceAttrSet("omada_wireless_network.lab", "id")},
   {ResourceName: "omada_wireless_network.lab", ImportState: true, ImportStateVerify: true, ImportStateVerifyIgnore: []string{"passphrase"}, ImportStateIdFunc: func(state *terraform.State) (string, error) {
    attrs := state.RootModule().Resources["omada_wireless_network.lab"].Primary.Attributes
    return attrs["site_id"] + "/" + attrs["wlan_group_id"] + "/" + attrs["id"], nil
   }},
   {Config: config("L1 Custom", custom), Check: resource.ComposeTestCheckFunc(resource.TestCheckResourceAttrPair("omada_wireless_network.lab", "wlan_group_id", "omada_wlan_group.lab", "id"), observe(false))},
   {Config: config("L1 Custom", custom), PlanOnly: true},
   {Config: config("L1 Updated", custom), Check: resource.ComposeTestCheckFunc(resource.TestCheckResourceAttr("omada_wireless_network.lab", "name", "L1 Updated"), observe(true))},
   {Config: config("L1 Updated", custom), PlanOnly: true},
  },
 })
}

func TestControllerLabSSIDUpdateVariants(t *testing.T) {
 if os.Getenv("OMADA_CONTROLLER_LAB") != "1" { t.Skip("disposable controller lab only") }
 if os.Getenv("OMADA_URL") != "https://controller:8043" { t.Fatal("refusing a target outside the isolated lab") }
 config := func(password string, vlan bool) string {
  setting := ""
  if vlan { setting = "vlan_id = omada_network.lab.vlan_id" }
  return fmt.Sprintf(`
resource "omada_network" "lab" {
 site_id = %q
 name = "L7 VLAN"
 purpose = "vlan"
 vlan_id = 197
}
resource "omada_wireless_network" "lab" {
 site_id = %q
 name = "L7 Variants"
 security = 3
 passphrase = %q
 pmf_mode = 2
 %s
}
`, os.Getenv("OMADA_TEST_SITE_ID"), os.Getenv("OMADA_TEST_SITE_ID"), password, setting)
 }
 var originalID string
 check := func(vlan string, enabled bool) resource.TestCheckFunc {
  return func(state *terraform.State) error {
   attrs := state.RootModule().Resources["omada_wireless_network.lab"].Primary.Attributes
   if originalID != "" && attrs["id"] != originalID { return fmt.Errorf("update unexpectedly replaced SSID") }
   originalID = attrs["id"]
   if attrs["vlan_id"] != vlan { return fmt.Errorf("unexpected VLAN state") }
   c, err := client.NewClient(os.Getenv("OMADA_URL"), os.Getenv("OMADA_USERNAME"), os.Getenv("OMADA_PASSWORD"), true)
   if err != nil { return err }
   raw, err := c.GetWirelessNetworkRaw(context.Background(), attrs["site_id"], attrs["wlan_group_id"], attrs["id"])
   if err != nil { return err }
   psk, ok := raw["pskSetting"].(map[string]any)
   if !ok || psk["securityKey"] != attrs["passphrase"] { return fmt.Errorf("controller credential readback did not match configured synthetic password") }
   if raw["vlanEnable"] != enabled { return fmt.Errorf("VLAN summary does not match authoritative setting") }
   return nil
  }
 }
 resource.Test(t, resource.TestCase{
  PreCheck: func() { testAccPreCheck(t) }, ProtoV6ProviderFactories: testAccProtoV6ProviderFactories,
  Steps: []resource.TestStep{
   {Config: config("L7-synthetic-password", false), Check: check("", false)},
   {Config: config("L7-rotated-password", false), Check: resource.ComposeTestCheckFunc(check("", false), resource.TestCheckResourceAttr("omada_wireless_network.lab", "passphrase", "L7-rotated-password"))},
   {Config: config("L7-rotated-password", true), Check: check("197", true)},
   {Config: config("L7-rotated-password", true), PlanOnly: true},
   {Config: config("L7-rotated-password", false), Check: check("", false)},
   {Config: config("L7-rotated-password", false), PlanOnly: true},
  },
 })
}

func TestControllerLabPMF(t *testing.T) {
 if os.Getenv("OMADA_CONTROLLER_LAB") != "1" { t.Skip("disposable controller lab only") }
 if os.Getenv("OMADA_URL") != "https://controller:8043" { t.Fatal("refusing a target outside the isolated lab") }
 config := func(name string, security int, pmf string) string {
  password := ""
  if security == 3 { password = `passphrase = "L6-synthetic-password"` }
  return fmt.Sprintf(`
resource "omada_wireless_network" "lab" {
 site_id = %q
 name = %q
 security = %d
 %s
 %s
}
`,os.Getenv("OMADA_TEST_SITE_ID"),name,security,password,pmf)
 }
 var id string
 check := func(security,pmf string) resource.TestCheckFunc {
  return resource.ComposeTestCheckFunc(
   resource.TestCheckResourceAttr("omada_wireless_network.lab","security",security),
   resource.TestCheckResourceAttr("omada_wireless_network.lab","pmf_mode",pmf),
   func(state *terraform.State) error {
    current := state.RootModule().Resources["omada_wireless_network.lab"].Primary.ID
    if id!="" && id!=current { return fmt.Errorf("security/PMF update replaced SSID") }
    id=current
    return nil
   },
  )
 }
 resource.Test(t,resource.TestCase{PreCheck:func(){testAccPreCheck(t)},ProtoV6ProviderFactories:testAccProtoV6ProviderFactories,
  Steps: []resource.TestStep{
   {Config:config("L6 Open",0,""),Check:check("0","0")},
   {Config:config("L6 Open",0,""),PlanOnly:true},
   {Config:config("L6 Renamed",0,""),Check:check("0","0")},
   {Config:config("L6 Renamed",3,""),Check:check("3","2")},
   {Config:config("L6 Renamed",3,"pmf_mode = 1"),Check:check("3","1")},
   {Config:config("L6 Renamed",3,"pmf_mode = 3"),Check:check("3","3")},
   {Config:config("L6 Renamed",3,"pmf_mode = 3"),PlanOnly:true},
   {Config:config("L6 Renamed",0,""),Check:check("0","0")},
   {Config:config("L6 Renamed",0,""),PlanOnly:true},
  },
 })
}
